2077: export function canStrike(s: GameState, attacker: Unit, target: Unit): boolean {
2078:   // 射程要按"这一发从哪打到哪"算：站丘陵上俯射多一格，往高处仰攻少一格，
2079:   // 站在林子里连平射都被树冠吃掉一格（见 rangeModAgainst）。
2080:   const range = effectiveRangeAgainst(s, attacker, tileAt(s, target.x, target.y));
2081:   // 地面打空中要多算一格（见 strikeDistance）；同格判定仍然看"真实相对位置"，
2082:   // 否则"空中压着地面"这个免费的第一刀会被 +1 顺手吃掉
2083:   const real = distBetweenUnits(s, attacker, target);
2084:   const d = strikeDistance(s, attacker, target);
2085:   if (d > range) return false;
2086:   // 距离 0 只会在"空中压着地面"时出现——那是贴着打，不是"打不到自己"
2087:   if (real === 0) return isAirborne(s, attacker) !== isAirborne(s, target);
2088:   return true;
2089: }
